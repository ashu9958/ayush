public class GreatestOfThree {

    public static void main(String[] args) {
        int num1 = 10;
        int num2 = 25;
        int num3 = 15;

        int greatest;

        if (num1 >= num2 && num1 >= num3) {
            greatest = num1;
        } else if (num2 >= num1 && num2 >= num3) {
            greatest = num2;
        } else {
            greatest = num3;
        }

        System.out.println("The greatest number is: " + greatest);
    }
}






junit test case:-

import static org.junit.Assert.*;
import org.junit.Test;

public class GreatestOfThreeTest {

    @Test
    public void testFindGreatest() {
        GreatestOfThree obj = new GreatestOfThree();

        assertEquals(25, obj.findGreatest(10, 25, 15));
        assertEquals(50, obj.findGreatest(50, 25, 15));
        assertEquals(30, obj.findGreatest(10, 30, 30));
        assertEquals(5, obj.findGreatest(5, 5, 5));
        assertEquals(-1, obj.findGreatest(-10, -20, -1));
    }
}
